    \end{itemize}
\end{itemize}

\subsection{RSA Verschlüsselung}
\subsubsection*{RSA Schlüsselerzeugung}
\begin{itemize}
    \item Schlüsselerzeugung \blue{GenRSA(n)} (Sicherheitsparameter n)
    \begin{itemize}
        \item Wähle zwei \blue{n-bit Primzahlen} \fatsf{p, q} mit \fatsf{p $\neq$ q}
        \item Berechne \blue{N = p $\times$ q}
        \item Wähle \blue{e > 1}, sodass \fatsf{ggT(e, $\varphi$(N)) = 1}
        \begin{defBox}
            Eulersche Phi-Funktion $\varphi$ = (p - 1)(q - 1) = Anzahl Elemente in $\mathbb{Z}^{*}_{N}$ (positive natürliche Zahlen < N, teilerfremd zu N) \red{(Ohne Wissen von p, q sehr schwer zu berechnen)}
        \end{defBox}
        \item Berechne \fatsf{d = $e^{-1}$ mod $\varphi$(N)}
        \item Ausgabe: \fatsf{(N,e,d)}
    \end{itemize}
    $\rightarrow$ \fatsf{(N,e)} ist Öffentlicher Schlüssel\\
    $\rightarrow$ \fatsf{(N,d)} ist Geheimer Schlüssel
\end{itemize}

\subsubsection*{RSA Annahme}
Um die Invertierung der RSA Funktion zu berechnen, benötigen wir d
\begin{itemize}
    \item Seien \fatsf{(N,e,d) = GenRSA(n)}
    \item Wähle y zufällig in $\mathbb{Z}^{*}_{N}$
    \item Gegeben (N,e,y), es ist \red{schwierig} x zu berechnen, so dass: \fatsf{y $\equiv$ $x_{e}$ mod N}
\end{itemize}
Möglicher Angriff auf RSA Annahme:
\begin{enumerate}
    \item Faktorisiere \fatsf{N = p $\times$ q}
    \item Berechne \fatsf{$\varphi$(N) = (p - 1) $\times$ (q - 1)}
    \item Berechne \fatsf{d = $e^{-1}$ mod $\varphi$(N)}
\end{enumerate}
$\rightarrow$ Schwierigkeit der \blue{Faktorisierung} ist notwendige \blue{Annahme} für Sicherheit des RSA Verfahrens

\subsubsection*{Wahl der RSA Parameter}
Ist die Faktorisierung von $3 \times 5 = 15$ schwer?

    \item Konkrete Schwierigkeit der Annahmen hängt von Parametern ab (z.B. Größe der Zahl (Gemessen in Bit-Länge (= log der Zahl)))

\subsubsection*{Einschub: Homomorphe Verschlüsselung}
Verschlüsselungsverfahren heißt \blue{(multiplikativ) homomorph}, wenn:
\fatsf{\[
    Enc(pk,m_{0}) \times Enc(pk,m_{1}) = Enc(pk,m_{0} \times m_{1})
\]}
(Produkt zweier verschlüsselter Nachrichten entspricht Verschlüsselung des Produkts)
\begin{itemize}
    \item Textbuch RSA ist homomorph:
    \[
        (m_{0}^{e} mod N) \times (m_{1}^{e} mod N) = (m_{0} \times m_{1})^{e} mod N
    \]
    \item Wir können Operationen auf den Klartext ausführen, ohne entschlüsseln zu müssen!
\end{itemize}
$\rightarrow$ Hauptsächlich Problem für IND-CCA Sicherheit
\subsubsection*{Sicherheit von Textbuch RSA}
Textbuch RSA Verschlüsselung ist \blue{nicht} IND-CPA sicher
\begin{itemize}
    \item Grund: Die Verschlüsselung ist \blue{deterministisch}
    \begin{defBox}
        Faust-Regel: \fatsf{Kein deterministisches PKE ist IND-CPA-sicher}
    \end{defBox}
\end{itemize}
Wie kann ein Angreifer das IND-CPA Spiel gewinnen?
\begin{enumerate}
    \item Wähle zwei Nachrichten $m_{0}, m_{1}$ (mit gleicher Länge)
    \item Berechne $c_{0} = Enc(pk, m_{0})$ mit Hilfe des öffentlichen Schlüssels
    \item Vergleiche den Challenge Chiffretext $c^{*}$ mit $c_{0}$
\end{enumerate}

\subsubsection*{Angriffe auf RSA-Basierte Chiffren}
\begin{itemize}

    \item Angriffe durch Ausnutzen von Padding Fehlern (Padding-Oracle-Attacks)

    \end{itemize}
    \item Angriffe bei kleinen Exponenten e bzw. Teilwissen über einen Primfaktor
    \begin{itemize}
        \item Beispiel: Coppersmith Angriff/Hastads Angriff mit chinesischem Restsatz
    \end{itemize}
    \item Und viele mehr...
    \item \fatsf{Erinnerung:} Vertraulichkeit $\neq$ Integrität $\neq$ Authentifizierung!
    \begin{itemize}
        \item Beispiel: Angriff auf Schlüsselmaterial in Nextcloud durch fehlende Authentifizierung von RSA-OAEP
        \item Vergleichbare Angriffe bei anderen Cloud-Storage Anbietern (Sync, pCloud, Tresorit)
    \end{itemize}
\end{itemize}

\subsubsection*{MACs VS. Signaturen}
\fatsf{Vorteile:}\\
\underline{MACs}
\begin{itemize}
    \item Signifikant \fatsf{schneller} und \fatsf{speichereffizienter}, da symmetrische Kryptoprimitive
\end{itemize}
\underline{Digitale Signaturen}
\begin{itemize}
    \item Ähnlich zu PKE: \fatsf{einfacheres Key Management}
    \item \fatsf{Öffentlich verifizierbar:} Eine Signatur (m,s) kann an andere Partei weitergeleitet werden
    \item \fatsf{Nicht Abstreitbarkeit:} Es ist eindeutig klar, wer die Nachricht signiert hat (bei MACs zwei Parteien)
\end{itemize}
\fatsf{Allgemeine Faustregel:}
\begin{itemize}
    \item MACs, wenn eine Person signiert und \red{eine} zweite Person verifiziert
    \item Signaturen, wenn eine Person signiert und \red{viele} Personen verifizieren
\end{itemize}
\fatsf{3 Arten für Datenintegrität}\\
Gegenüberstellung am Beispiel Softwareupdates:
\begin{enumerate}
    \item \fatsf{Kollisions-resistente Hashfunktion}
    \begin{itemize}
        \item Benötigt einen öffentlichen \blue{schreibgeschützten} Bereich (z.B. Webseite)
    \end{itemize}
    \item \fatsf{Digitale Signaturen}
    \begin{itemize}
        \item Hersteller muss über langen Zeitraum einen \blue{privaten Schlüssel aufbewahren}
        \item Download-Seite muss nicht vertraut werden
    \end{itemize}
    \item \fatsf{MACs}
    \begin{itemize}
        \item Hersteller muss über langen Zeitraum \blue{Geheimnis aufbewahren} (für die Erstellung von MAC Schlüssel pro Benutzer)
        \item Hersteller muss \blue{separaten MAC} der Software \blue{für jeden Benutzer} erstellen
    \end{itemize}
\end{enumerate}

\subsubsection*{Risiken schwacher Zufallsgenerierung}
\begin{itemize}
    \item Was kann schiefgehen?
    \begin{itemize}
        \item \blue{Vorhersehbarkeit} $\rightarrow$ Angreifer kann zufünftige \enquote{Zufalls} Werte berechnen
        \item \blue{Geringe Entropie} $\rightarrow$ Brute Force Angriff wegen geringer Anzahl möglicher \enquote{Zufalls} Werte
        \item \blue{Wiederhalbare Muster} $\rightarrow$ Schwache Zufallsgeneratoren können zyklische Muster erzeugen
    \end{itemize}
    \item Praktische Beispiele
    \begin{itemize}
        \item Debian OpenSSL Bug (2008)
        \item Bitcoin Wallet Hack (2013)
        \item Nextcloud Weak Passwort Generation (2023)
    \end{itemize}
\end{itemize}
\subsubsection*{Quellen für Zufälligkeit}
\begin{itemize}
    \item \fatsf{Pseudozufallszahlengeneratoren (PRNGs)}
    \begin{itemize}
        \item \fatsf{Beschreibung:} \blue{Scheinbar zufällige} Zahlen auf Basis eines Anfangswertes (Seed)
        \item \fatsf{Eigenschaften:} Schnell, deterministisch, wiederholbar
        \item \fatsf{Anwendung:} In Software, wenn \blue{Geschwindigkeit \& Effizienz} im Vordergrund stehen
        \item \fatsf{Risiko:} \blue{Vorhersehbar}, wenn Seed bekannt oder schwach\\
        $\rightarrow$ Für Kryptographie und Passwörter nutze kryptographischen PRNG
    \end{itemize}
    \item \fatsf{Echte Zufallszahlengeneratoren (TRNGs)}
    \begin{itemize}
        \item \fatsf{Beschreibung:} \blue{Echte Zufallszahlen} ausphysikalischen Prozessen
        \item \fatsf{Eigenschaften:} Hohe \blue{Unvorhersehbarkeit}, langsam, oft teurer in der Umsetzung
        \item \fatsf{Beispiele:} Halbleiterrauschen, Atmosphärischen Rauschen, radioaktiver Zufall
        \item \fatsf{Anwendung:} Kritische Sicherheitssysteme, z.B. in Hardware-Sicherheitsmodulen (HSMs)
    \end{itemize}
\end{itemize}
